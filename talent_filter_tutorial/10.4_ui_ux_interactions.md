# 10.4 UI/UX: Interactions

## Understanding User Interactions

In this section, we'll explore the interactive elements and JavaScript functionality in the Talent Filter application. These interactions enhance the user experience by making the application more dynamic and responsive.

### Main JavaScript (main.js)

The `main.js` file contains common functionality used across the application:

```javascript
// main.js

// Document ready function
document.addEventListener('DOMContentLoaded', function() {
    // Initialize mobile navigation
    initMobileNav();
    
    // Initialize notifications
    initNotifications();
    
    // Initialize modals
    initModals();
    
    // Initialize delete buttons
    initDeleteButtons();
    
    // Initialize file inputs
    initFileInputs();
});

// Initialize mobile navigation
function initMobileNav() {
    // Get all "navbar-burger" elements
    const $navbarBurgers = Array.prototype.slice.call(document.querySelectorAll('.navbar-burger'), 0);
    
    // Check if there are any navbar burgers
    if ($navbarBurgers.length > 0) {
        // Add a click event on each of them
        $navbarBurgers.forEach(el => {
            el.addEventListener('click', () => {
                // Get the target from the "data-target" attribute
                const target = el.dataset.target;
                const $target = document.getElementById(target);
                
                // Toggle the "is-active" class on both the "navbar-burger" and the "navbar-menu"
                el.classList.toggle('is-active');
                $target.classList.toggle('is-active');
            });
        });
    }
}

// Initialize notifications
function initNotifications() {
    // Auto-hide notifications after 5 seconds
    const notifications = document.querySelectorAll('.notification:not(.is-permanent)');
    notifications.forEach(notification => {
        setTimeout(() => {
            notification.style.transition = 'opacity 0.5s ease-out';
            notification.style.opacity = '0';
            setTimeout(() => {
                notification.remove();
            }, 500);
        }, 5000);
    });
    
    // Add click event to delete buttons
    const deleteButtons = document.querySelectorAll('.notification .delete');
    deleteButtons.forEach(button => {
        button.addEventListener('click', () => {
            const notification = button.parentNode;
            notification.style.transition = 'opacity 0.5s ease-out';
            notification.style.opacity = '0';
            setTimeout(() => {
                notification.remove();
            }, 500);
        });
    });
}

// Initialize modals
function initModals() {
    // Functions to open and close a modal
    function openModal($el) {
        $el.classList.add('is-active');
        document.documentElement.classList.add('is-clipped');
    }

    function closeModal($el) {
        $el.classList.remove('is-active');
        document.documentElement.classList.remove('is-clipped');
    }

    function closeAllModals() {
        (document.querySelectorAll('.modal') || []).forEach(($modal) => {
            closeModal($modal);
        });
    }

    // Add a click event on buttons to open a specific modal
    (document.querySelectorAll('.js-modal-trigger') || []).forEach(($trigger) => {
        const modal = $trigger.dataset.target;
        const $target = document.getElementById(modal);

        $trigger.addEventListener('click', () => {
            openModal($target);
        });
    });

    // Add a click event on various child elements to close the parent modal
    (document.querySelectorAll('.modal-background, .modal-close, .modal-card-head .delete, .modal-card-foot .button') || []).forEach(($close) => {
        const $target = $close.closest('.modal');

        $close.addEventListener('click', () => {
            closeModal($target);
        });
    });

    // Add a keyboard event to close all modals
    document.addEventListener('keydown', (event) => {
        if (event.code === 'Escape') {
            closeAllModals();
        }
    });
}

// Initialize delete buttons
function initDeleteButtons() {
    // Add click event to delete buttons
    const deleteButtons = document.querySelectorAll('.delete:not(.notification .delete)');
    deleteButtons.forEach(button => {
        button.addEventListener('click', () => {
            const parent = button.parentNode;
            parent.remove();
        });
    });
}

// Initialize file inputs
function initFileInputs() {
    // Add change event to file inputs
    const fileInputs = document.querySelectorAll('.file-input');
    fileInputs.forEach(input => {
        input.addEventListener('change', () => {
            const fileName = input.files[0] ? input.files[0].name : 'No file selected';
            const fileNameElement = input.parentNode.querySelector('.file-name');
            if (fileNameElement) {
                fileNameElement.textContent = fileName;
            }
        });
    });
}

// Utility function to get CSRF token
function getCsrfToken() {
    return document.querySelector('meta[name="csrf-token"]').getAttribute('content');
}

// Utility function to format date
function formatDate(dateString) {
    const date = new Date(dateString);
    return date.toLocaleDateString('en-US', {
        year: 'numeric',
        month: 'short',
        day: 'numeric'
    });
}

// Utility function to show a notification
function showNotification(message, type = 'info') {
    const notification = document.createElement('div');
    notification.className = `notification is-${type}`;
    notification.innerHTML = `
        <button class="delete"></button>
        ${message}
    `;
    
    const container = document.querySelector('.notification-container') || document.body;
    container.appendChild(notification);
    
    // Initialize the notification
    initNotifications();
}
```

## Form Interactions

### Form Validation

The application includes client-side form validation:

```javascript
// Validate form on submit
document.querySelectorAll('form').forEach(form => {
    form.addEventListener('submit', function(event) {
        let isValid = true;
        
        // Check required fields
        const requiredFields = form.querySelectorAll('[required]');
        requiredFields.forEach(field => {
            if (!field.value.trim()) {
                isValid = false;
                field.classList.add('is-danger');
                
                // Add error message if it doesn't exist
                let errorMessage = field.parentNode.querySelector('.help.is-danger');
                if (!errorMessage) {
                    errorMessage = document.createElement('p');
                    errorMessage.className = 'help is-danger';
                    errorMessage.textContent = 'This field is required';
                    field.parentNode.appendChild(errorMessage);
                }
            } else {
                field.classList.remove('is-danger');
                
                // Remove error message if it exists
                const errorMessage = field.parentNode.querySelector('.help.is-danger');
                if (errorMessage) {
                    errorMessage.remove();
                }
            }
        });
        
        // Check email fields
        const emailFields = form.querySelectorAll('input[type="email"]');
        emailFields.forEach(field => {
            if (field.value.trim() && !isValidEmail(field.value.trim())) {
                isValid = false;
                field.classList.add('is-danger');
                
                // Add error message if it doesn't exist
                let errorMessage = field.parentNode.querySelector('.help.is-danger');
                if (!errorMessage) {
                    errorMessage = document.createElement('p');
                    errorMessage.className = 'help is-danger';
                    errorMessage.textContent = 'Please enter a valid email address';
                    field.parentNode.appendChild(errorMessage);
                }
            }
        });
        
        if (!isValid) {
            event.preventDefault();
        }
    });
});

// Validate email format
function isValidEmail(email) {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return emailRegex.test(email);
}
```

### Tag Input

The application includes a custom tag input component:

```javascript
// Tag input functionality
document.addEventListener('DOMContentLoaded', function() {
    // Add tag when Enter is pressed
    document.querySelectorAll('.tag-input-field-input').forEach(input => {
        input.addEventListener('keydown', function(e) {
            if (e.key === 'Enter') {
                e.preventDefault();
                
                const tag = this.value.trim();
                if (tag) {
                    // Add the tag
                    const tagsContainer = this.closest('.tag-input-field').querySelector('.tags-container');
                    const tagElement = document.createElement('span');
                    tagElement.className = 'tag is-primary';
                    tagElement.innerHTML = `${tag}<button type="button" class="delete is-small"></button>`;
                    tagsContainer.appendChild(tagElement);
                    
                    // Clear the input
                    this.value = '';
                    
                    // Update the hidden input
                    updateTagsInput(this.closest('.tag-input-container'));
                }
            }
        });
    });
    
    // Remove tag when delete button is clicked
    document.addEventListener('click', function(e) {
        if (e.target.matches('.tag .delete')) {
            e.target.parentNode.remove();
            updateTagsInput(e.target.closest('.tag-input-container'));
        }
    });
    
    // Update the hidden input with all tags
    function updateTagsInput(container) {
        const tags = [];
        container.querySelectorAll('.tag').forEach(tag => {
            // Get the text content without the delete button text
            const tagText = tag.childNodes[0].nodeValue.trim();
            tags.push(tagText);
        });
        container.querySelector('input[type="hidden"]').value = tags.join(',');
    }
});
```

## AJAX Interactions

### Job Match Analysis

The application uses AJAX to analyze job matches:

```javascript
// Analyze job match
document.getElementById('analyze-match-btn').addEventListener('click', function() {
    const jobId = this.dataset.jobId;
    const resultContainer = document.getElementById('match-result-container');
    const loadingContainer = document.getElementById('match-loading-container');
    
    // Show loading state
    resultContainer.classList.add('is-hidden');
    loadingContainer.classList.remove('is-hidden');
    
    // Start the analysis
    fetch(`/jobs/${jobId}/analyze-match/`, {
        method: 'POST',
        headers: {
            'X-CSRFToken': getCsrfToken(),
            'Content-Type': 'application/json'
        }
    })
    .then(response => response.json())
    .then(data => {
        if (data.status === 'processing') {
            // Start polling for results
            pollMatchResults(jobId);
        } else if (data.status === 'error') {
            // Show error message
            showNotification(data.message, 'danger');
            loadingContainer.classList.add('is-hidden');
        }
    })
    .catch(error => {
        console.error('Error:', error);
        showNotification('An error occurred while analyzing the match.', 'danger');
        loadingContainer.classList.add('is-hidden');
    });
});

// Poll for match results
function pollMatchResults(jobId) {
    const resultContainer = document.getElementById('match-result-container');
    const loadingContainer = document.getElementById('match-loading-container');
    
    const pollInterval = setInterval(() => {
        fetch(`/jobs/${jobId}/analyze-match/?check_status=true`)
        .then(response => response.json())
        .then(data => {
            if (data.status === 'success') {
                // Show results
                clearInterval(pollInterval);
                displayMatchResults(data.result);
                resultContainer.classList.remove('is-hidden');
                loadingContainer.classList.add('is-hidden');
            } else if (data.status === 'error') {
                // Show error message
                clearInterval(pollInterval);
                showNotification(data.message, 'danger');
                loadingContainer.classList.add('is-hidden');
            }
            // If still processing, continue polling
        })
        .catch(error => {
            console.error('Error:', error);
            clearInterval(pollInterval);
            showNotification('An error occurred while checking match results.', 'danger');
            loadingContainer.classList.add('is-hidden');
        });
    }, 2000); // Poll every 2 seconds
}

// Display match results
function displayMatchResults(result) {
    const resultContainer = document.getElementById('match-result-container');
    
    // Update match score
    const scoreElement = resultContainer.querySelector('.match-score');
    scoreElement.textContent = result.match_score;
    
    // Update progress bar
    const progressBar = resultContainer.querySelector('.progress');
    progressBar.value = result.match_score;
    
    // Update matching skills
    const matchingSkillsList = resultContainer.querySelector('.matching-skills');
    matchingSkillsList.innerHTML = '';
    result.matching_skills.forEach(skill => {
        const li = document.createElement('li');
        li.innerHTML = `<span class="icon has-text-success"><i class="fas fa-check"></i></span> ${skill}`;
        matchingSkillsList.appendChild(li);
    });
    
    // Update missing skills
    const missingSkillsList = resultContainer.querySelector('.missing-skills');
    missingSkillsList.innerHTML = '';
    result.missing_skills.forEach(skill => {
        const li = document.createElement('li');
        li.innerHTML = `<span class="icon has-text-danger"><i class="fas fa-times"></i></span> ${skill}`;
        missingSkillsList.appendChild(li);
    });
    
    // Update assessment sections
    resultContainer.querySelector('.experience-match').textContent = result.experience_match;
    resultContainer.querySelector('.education-match').textContent = result.education_match;
    resultContainer.querySelector('.overall-assessment').textContent = result.overall_assessment;
    resultContainer.querySelector('.improvement-suggestions').textContent = result.improvement_suggestions;
}
```

### Resume Extraction

The application uses AJAX to extract resume data:

```javascript
// Extract resume data
document.getElementById('extract-resume-btn').addEventListener('click', function() {
    const extractionContainer = document.getElementById('extraction-container');
    const progressBar = document.getElementById('extraction-progress');
    const statusText = document.getElementById('extraction-status');
    
    // Show the extraction progress container
    extractionContainer.classList.remove('is-hidden');
    
    // Reset progress bar and status
    progressBar.style.width = '0%';
    statusText.textContent = 'Starting extraction...';
    
    // Disable the form while extraction is in progress
    document.querySelectorAll('#profile-form :input').forEach(input => {
        input.disabled = true;
    });
    
    // Send the extraction request
    fetch('/job-seeker-profile/extract-resume/', {
        method: 'POST',
        headers: {
            'X-Requested-With': 'XMLHttpRequest',
            'X-CSRFToken': getCsrfToken()
        }
    })
    .then(response => response.json())
    .then(data => {
        // Initial response, status checking will handle the rest
        checkExtractionStatus();
    })
    .catch(error => {
        console.error('Error:', error);
        statusText.textContent = 'Error starting extraction';
        
        // Enable the form
        document.querySelectorAll('#profile-form :input').forEach(input => {
            input.disabled = false;
        });
    });
});

// Check extraction status
function checkExtractionStatus() {
    const extractionContainer = document.getElementById('extraction-container');
    const progressBar = document.getElementById('extraction-progress');
    const statusText = document.getElementById('extraction-status');
    
    fetch('/job-seeker-profile/extract-resume/', {
        method: 'GET',
        headers: {
            'X-Requested-With': 'XMLHttpRequest'
        }
    })
    .then(response => response.json())
    .then(data => {
        // Update progress bar
        progressBar.style.width = data.progress + '%';
        
        // Update status message
        statusText.textContent = data.message;
        
        if (data.overall_status === 'completed') {
            // Extraction completed successfully
            extractionContainer.classList.add('is-success');
            
            // Update profile fields with extracted data
            if (data.extracted_data) {
                if (data.extracted_data.skills) {
                    document.getElementById('id_skills').value = data.extracted_data.skills;
                }
                if (data.extracted_data.education) {
                    document.getElementById('id_education').value = data.extracted_data.education;
                }
                if (data.extracted_data.experience_years) {
                    document.getElementById('id_experience_years').value = data.extracted_data.experience_years;
                }
                if (data.extracted_data.location) {
                    document.getElementById('id_location').value = data.extracted_data.location;
                }
            }
            
            // Show success message
            showNotification('Resume data extracted successfully!', 'success');
            
            // Enable the form
            document.querySelectorAll('#profile-form :input').forEach(input => {
                input.disabled = false;
            });
            
        } else if (data.overall_status === 'error') {
            // Extraction failed
            extractionContainer.classList.add('is-danger');
            
            // Show error message
            showNotification('Failed to extract resume data: ' + data.message, 'danger');
            
            // Enable the form
            document.querySelectorAll('#profile-form :input').forEach(input => {
                input.disabled = false;
            });
            
        } else {
            // Still processing, continue checking
            setTimeout(checkExtractionStatus, 2000);
        }
    })
    .catch(error => {
        console.error('Error:', error);
        statusText.textContent = 'Error checking extraction status';
        
        // Enable the form
        document.querySelectorAll('#profile-form :input').forEach(input => {
            input.disabled = false;
        });
    });
}
```

### Notification Handling

The application uses AJAX to handle notifications:

```javascript
// Mark notification as read
document.querySelectorAll('.notification-item').forEach(item => {
    item.addEventListener('click', function(e) {
        e.preventDefault();
        
        const notificationId = this.dataset.notificationId;
        const url = this.getAttribute('href');
        
        fetch(`/notifications/${notificationId}/mark-read/`, {
            method: 'POST',
            headers: {
                'X-CSRFToken': getCsrfToken(),
                'X-Requested-With': 'XMLHttpRequest'
            }
        })
        .then(response => response.json())
        .then(data => {
            // If there's a related link, redirect to it
            if (data.related_link) {
                window.location.href = data.related_link;
            } else {
                // Otherwise, update the UI
                updateNotificationCount();
                this.classList.remove('unread');
            }
        })
        .catch(error => {
            console.error('Error:', error);
        });
    });
});

// Mark all notifications as read
document.querySelector('.mark-all-read').addEventListener('click', function(e) {
    e.preventDefault();
    
    fetch('/notifications/mark-all-read/', {
        method: 'POST',

    })
    .then(response => response.json())
    .then(data => {
        // Update the UI

        document.querySelectorAll('.notification-item').forEach(item => {
            item.classList.remove('unread');
        });
    })
    .catch(error => {
        console.error('Error:', error);
    });

// Update notification count
function updateNotificationCount() {
    fetch('/notifications/count/', {
        method: 'GET',
        headers: {
            'X-Requested-With': 'XMLHttpRequest'
        }
    })
    .then(response => response.json())
    .then(data => {
        const count = data.count;
        const countElement = document.querySelector('.notification-count');
        
        if (count > 0) {
            countElement.textContent = count;
            countElement.classList.remove('is-hidden');
        } else {
            countElement.classList.add('is-hidden');
        }
    })
    .catch(error => {
        console.error('Error:', error);
    });
}

// Poll for new notifications every 60 seconds
setInterval(updateNotificationCount, 60000);
```

### Filters

The application includes interactive filters:

```javascript
// Filter functionality
document.getElementById('apply-filters').addEventListener('click', function() {
    const statusFilter = document.getElementById('status-filter').value;
    const jobFilter = document.getElementById('job-filter').value;
    const searchQuery = document.getElementById('search-input').value;
    
    // Build the query string
    let queryString = '';
    if (statusFilter) {
        queryString += `status=${encodeURIComponent(statusFilter)}&`;
    }
    if (jobFilter) {
        queryString += `job=${encodeURIComponent(jobFilter)}&`;
    }
    if (searchQuery) {
        queryString += `search=${encodeURIComponent(searchQuery)}&`;
    }
    
    // Remove trailing ampersand
    if (queryString.endsWith('&')) {
        queryString = queryString.slice(0, -1);
    }
    
    // Redirect to the filtered URL
    window.location.href = `?${queryString}`;
});

// Handle Enter key in search input
document.getElementById('search-input').addEventListener('keydown', function(e) {
    if (e.key === 'Enter') {
        e.preventDefault();
        document.getElementById('apply-filters').click();
    }
});
```
