# 6.1 AI Integration: Resume Parsing

## Understanding AI-Powered Resume Parsing

In this section, we'll explore how Talent Filter uses AI to extract information from resumes. This feature helps job seekers automatically populate their profiles with skills, education, experience, and location information.

## Overview of Resume Parsing

Resume parsing is the process of extracting structured information from unstructured resume documents. Talent Filter uses the Google Gemini API to analyze resume text and extract relevant information. The process involves:

1. Reading the resume file (PDF or DOCX)
2. Extracting text content
3. Sending the text to the Google Gemini API with specific prompts
4. Processing the API response
5. Updating the user's profile with the extracted information

## Reading Resume Files

The first step is to read the content of the uploaded resume file:

```python
def read_resume_file(file_path):
    """
    Read the content of a resume file (PDF or DOCX) and return it as a string.

    Args:
        file_path (str): Path to the resume file

    Returns:
        str: Content of the resume file as a string
    """
    file_ext = os.path.splitext(file_path)[1].lower()

    try:
        if file_ext == '.pdf':
            if PyPDF2 is None:
                logger.error("PyPDF2 is not installed. Cannot read PDF files.")
                return "Error: PyPDF2 is not installed. Cannot read PDF files."

            text = ""
            with open(file_path, 'rb') as file:
                pdf_reader = PyPDF2.PdfReader(file)
                for page_num in range(len(pdf_reader.pages)):
                    page = pdf_reader.pages[page_num]
                    text += page.extract_text()
            return text

        elif file_ext in ['.doc', '.docx']:
            if docx is None:
                logger.error("python-docx is not installed. Cannot read DOCX files.")
                return "Error: python-docx is not installed. Cannot read DOCX files."

            doc = docx.Document(file_path)
            text = [paragraph.text for paragraph in doc.paragraphs]
            return '\n'.join(text)

        else:
            logger.error(f"Unsupported file format: {file_ext}")
            return f"Error: Unsupported file format: {file_ext}"

    except Exception as e:
        logger.error(f"Error reading resume file: {str(e)}")
        return f"Error: {str(e)}"
```

## Chunking Text for API Processing

For large resumes, the text is split into manageable chunks:

```python
def split_text_into_chunks(text, chunk_size=200, overlap=30):
    """
    Split text into chunks of specified size with overlap.

    Args:
        text (str): The text to split
        chunk_size (int): The size of each chunk in words
        overlap (int): The number of words to overlap between chunks

    Returns:
        list: List of text chunks
    """
    # Split the text into words
    words = text.split()

    # If the text is smaller than the chunk size, return it as is
    if len(words) <= chunk_size:
        return [text]

    chunks = []
    start = 0

    while start < len(words):
        # Calculate end position for this chunk
        end = min(start + chunk_size, len(words))

        # Create the chunk from the words
        chunk = ' '.join(words[start:end])
        chunks.append(chunk)

        # Move the start position for the next chunk, accounting for overlap
        start = end - overlap if end < len(words) else len(words)

    return chunks
```

def make_api_call(prompt, max_retries=0, retry_delay=0):
    """
    Make a single API call to Google Gemini API with the given prompt.
    Returns None if the call fails.

    Args:
        prompt (str): The prompt to send to the API
        max_retries (int): Not used, kept for compatibility
        retry_delay (int): Not used, kept for compatibility

    Returns:
        dict or None: The parsed API response or None if failed
    """
    from requests.exceptions import RequestException
    from django.conf import settings
    import time

    try:
        # Log the API call attempt with timestamp
        timestamp = time.strftime("%Y-%m-%d %H:%M:%S")
        logger.info(f"[{timestamp}] GEMINI API CALL STARTED - Prompt length: {len(prompt)} characters")

        # Get API settings from Django settings
        api_key = settings.GEMINI_API_KEY
        api_url = settings.GEMINI_API_URL

        # Add API key to URL
        api_endpoint = f"{api_url}?key={api_key}"
        logger.info(f"[{timestamp}] Using Gemini API endpoint: {api_url}")

        # Prepare the request payload for Gemini API - using the correct format
        payload = {
            "contents": [{
                "parts": [{
                    "text": prompt
                }]
            }]
        }

        logger.info(f"[{timestamp}] Sending request to Gemini API...")

        # Make the API call with a timeout
        start_time = time.time()
        response = requests.post(api_endpoint, json=payload, timeout=30)

        # Log the response time
        elapsed_time = time.time() - start_time
        logger.info(f"[{timestamp}] GEMINI API RESPONSE RECEIVED - Time: {elapsed_time:.2f}s, Status: {response.status_code}")

        # If response is not successful, log and return None
        if response.status_code != 200:
            logger.error(f"Gemini API call failed with status code {response.status_code}: {response.text}")
            return None

        # Parse the API response
        api_response = response.json()

        # Log the raw response for debugging
        logger.debug(f"Gemini API raw response: {api_response}")

        # Extract the text from the response - Gemini 2.0 format
        if 'candidates' in api_response and api_response['candidates']:
            candidate = api_response['candidates'][0]
            if 'content' in candidate and 'parts' in candidate['content'] and candidate['content']['parts']:
                response_text = candidate['content']['parts'][0]['text']

                logger.info(f"[{timestamp}] Received response text: {response_text[:100]}...")

                # Extract JSON from the response text
                json_str = None

                # First try to find JSON in code blocks
                json_match = re.search(r'```json\n(.+?)\n```', response_text, re.DOTALL)
                if json_match:
                    json_str = json_match.group(1)
                    logger.info(f"[{timestamp}] Found JSON in code block")
                else:
                    # Try to find JSON object without code blocks
                    json_match = re.search(r'\{.+\}', response_text, re.DOTALL)
                    if json_match:
                        json_str = json_match.group(0)
                        logger.info(f"[{timestamp}] Found JSON object without code block")
                    else:
                        # Try to find JSON array without code blocks
                        json_match = re.search(r'\[.+\]', response_text, re.DOTALL)
                        if json_match:
                            json_str = json_match.group(0)
                            logger.info(f"[{timestamp}] Found JSON array without code block")
                        else:
                            # If the response is already a clean JSON array format (without brackets)
                            # This handles cases where the model returns just the array items
                            if response_text.strip().startswith('"') and response_text.strip().endswith('"'):
                                try:
                                    # Try to parse as a JSON array by adding brackets
                                    json_str = f"[{response_text}]"
                                    logger.info(f"[{timestamp}] Attempting to parse as JSON array items")
                                except:
                                    pass

                            if not json_str:
                                logger.error("Could not find JSON in Gemini API response")

                    json_str = json_str.replace('```json', '').replace('```', '').strip()

                    logger.info(f"[{timestamp}] Successfully parsed JSON")
                    return parsed_json
                except json.JSONDecodeError as e:
                    logger.error(f"Error parsing JSON: {str(e)}")
                    logger.error(f"JSON string that failed to parse: {json_str}")

                    # Try one simple cleanup attempt
                    try:
                        logger.info(f"[{timestamp}] Attempting JSON cleanup and retry...")
                        # Replace single quotes with double quotes
                        cleaned_str = json_str.replace("'", '"')
                        # Remove newline characters and extra spaces
                        cleaned_str = re.sub(r'\s*\n\s*', '', cleaned_str)
                        # Fix common JSON formatting issues
                        cleaned_str = cleaned_str.replace('\\n', '')

                        logger.info(f"[{timestamp}] Cleaned JSON: {cleaned_str[:100]}...")
                        parsed_json = json.loads(cleaned_str)
                        logger.info(f"[{timestamp}] Successfully parsed JSON after cleanup")
                        return parsed_json

                return None
        else:
            logger.error("No candidates in Gemini API response")
            return None

    except RequestException as e:
        # Handle network-related exceptions
        logger.error(f"Request exception: {str(e)}")
        return None

## Extraction Process

The main extraction function coordinates the process of extracting different types of information:

```python
def extract_resume_data_from_api(resume_file_path, user_id=None):
    """
    Call an external API to extract data from a resume file.

    Args:
        resume_file_path (str): Path to the resume file
        user_id (int, optional): User ID for tracking extraction status

    Returns:
        dict: Extracted data with keys 'skills', 'education', 'experience_years', 'location'
        or None if extraction failed
    """
    try:
        # Reset extraction status if user_id is provided
        if user_id:
            reset_extraction_status(user_id)
            update_extraction_status(
                user_id,
                overall_status=ExtractionStatus.ACTIVE,
                message="Starting resume extraction process..."
            )

        # Step 1: Reading the resume file
        if user_id:
            update_extraction_status(
                user_id,
                step=ExtractionStep.READING,
                status=ExtractionStatus.ACTIVE,
                message="Reading resume file..."
            )

        # Get the absolute file path
        abs_file_path = os.path.join(settings.MEDIA_ROOT, resume_file_path)

        # Check if file exists
        if not os.path.exists(abs_file_path):
            logger.error(f"Resume file not found at {abs_file_path}")
            if user_id:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.READING,
                    status=ExtractionStatus.ERROR,
                    message="Resume file not found"
                )
            return None

        # Read the resume file content
        resume_text = read_resume_file(abs_file_path)
        if resume_text.startswith("Error:"):
            logger.error(resume_text)
            if user_id:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.READING,
                    status=ExtractionStatus.ERROR,
                    message=resume_text
                )
            return None

        # Mark reading step as completed
        if user_id:
            update_extraction_status(
                user_id,
                step=ExtractionStep.READING,
                status=ExtractionStatus.COMPLETED,
                message="Resume file read successfully"
            )

        # Step 2: Extracting skills
        if user_id:
            update_extraction_status(
                user_id,
                step=ExtractionStep.SKILLS,
                status=ExtractionStatus.ACTIVE,
                message="Extracting skills from resume..."
            )

        # Call the API with the user_id and step for better status tracking
        skills_result = call_api_with_prompt(
            resume_text,
            SKILLS_EXTRACTION_PROMPT,
            user_id=user_id,
            step=ExtractionStep.SKILLS if user_id else None
        )

        # Convert the result to a comma-separated string
        skills = ""
        if skills_result and isinstance(skills_result, list):
            skills = ", ".join([str(skill).strip() for skill in skills_result if skill])

        if user_id:
            if skills:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.SKILLS,
                    status=ExtractionStatus.COMPLETED,
                    message=f"Extracted {len(skills.split(','))} skills"
                )
            else:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.SKILLS,
                    status=ExtractionStatus.ERROR,
                    message="Failed to extract skills"
                )

        # Step 3: Extracting education
        if user_id:
            update_extraction_status(
                user_id,
                step=ExtractionStep.EDUCATION,
                status=ExtractionStatus.ACTIVE,
                message="Extracting education information..."
            )

        # Call the API with the user_id and step for better status tracking
        education_result = call_api_with_prompt(
            resume_text,
            EDUCATION_EXTRACTION_PROMPT,
            user_id=user_id,
            step=ExtractionStep.EDUCATION if user_id else None
        )

        # Convert the result to a newline-separated string
        education = ""
        if education_result and isinstance(education_result, list):
            education = "\n".join([str(edu).strip() for edu in education_result if edu])

        if user_id:
            if education:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.EDUCATION,
                    status=ExtractionStatus.COMPLETED,
                    message=f"Extracted {len(education.split('\\n'))} education entries"
                )
            else:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.EDUCATION,
                    status=ExtractionStatus.ERROR,
                    message="Failed to extract education information"
                )

        # Step 4: Extracting experience
        if user_id:
            update_extraction_status(
                user_id,
                step=ExtractionStep.EXPERIENCE,
                status=ExtractionStatus.ACTIVE,
                message="Extracting work experience..."
            )

        # Call the API with the user_id and step for better status tracking
        experience_entries = call_api_with_prompt(
            resume_text,
            EXPERIENCE_EXTRACTION_PROMPT,
            user_id=user_id,
            step=ExtractionStep.EXPERIENCE if user_id else None
        )

        # Ensure the result is a list
        if not experience_entries or not isinstance(experience_entries, list):
            experience_entries = []
        else:
            experience_entries = [str(exp).strip() for exp in experience_entries if exp]

        if user_id:
            if experience_entries:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.EXPERIENCE,
                    status=ExtractionStatus.COMPLETED,
                    message=f"Extracted {len(experience_entries)} experience entries"
                )
            else:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.EXPERIENCE,
                    status=ExtractionStatus.ERROR,
                    message="Failed to extract work experience"
                )

        # Step 5: Extracting location
        if user_id:
            update_extraction_status(
                user_id,
                step=ExtractionStep.LOCATION,
                status=ExtractionStatus.ACTIVE,
                message="Extracting location information..."
            )

        # Call the API with the user_id and step for better status tracking
        location_result = call_api_with_prompt(
            resume_text,
            LOCATION_EXTRACTION_PROMPT,
            user_id=user_id,
            step=ExtractionStep.LOCATION if user_id else None
        )

        # Convert the result to a comma-separated string
        location = ""
        if location_result and isinstance(location_result, list):
            location = ", ".join([str(loc).strip() for loc in location_result if loc])

        if user_id:
            if location:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.LOCATION,
                    status=ExtractionStatus.COMPLETED,
                    message=f"Extracted location: {location}"
                )
            else:
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.LOCATION,
                    status=ExtractionStatus.ERROR,
                    message="Failed to extract location information"
                )

        # Calculate experience years
        experience_years = estimate_experience_years(experience_entries)

        # Step 6: Saving the data
        if user_id:
            update_extraction_status(
                user_id,
                step=ExtractionStep.SAVING,
                status=ExtractionStatus.ACTIVE,
                message="Preparing extracted data..."
            )

        # Combine the results
        processed_data = {
            'skills': skills,
            'education': education,
            'experience_years': experience_years,
            'location': location
        }

        # Check if we have at least some data
        has_some_data = any(processed_data.values())

        if user_id:
            if has_some_data:
                # First, mark the saving step as completed
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.SAVING,
                    status=ExtractionStatus.COMPLETED,
                    message="Data extraction completed successfully"
                )

                # Then, in a separate call, update the overall status to completed
                # This ensures the frontend can detect the completion
                update_extraction_status(
                    user_id,
                    message="Data extraction completed successfully",
                    extracted_data=processed_data,
                    overall_status=ExtractionStatus.COMPLETED,
                    progress=100  # Ensure progress is set to 100%
                )
            else:
                # Mark the saving step as error
                update_extraction_status(
                    user_id,
                    step=ExtractionStep.SAVING,
                    status=ExtractionStatus.ERROR,
                    message="No data could be extracted from the resume"
                )

                # Then update the overall status to error
                update_extraction_status(
                    user_id,
                    message="No data could be extracted from the resume",
                    overall_status=ExtractionStatus.ERROR
                )
                return None

        return processed_data

    except Exception as e:
        logger.error(f"Error extracting resume data: {str(e)}")
        if user_id:
            # Update the overall status to error
            update_extraction_status(
                user_id,
                overall_status=ExtractionStatus.ERROR,
                message=f"Error extracting resume data: {str(e)}",
                progress=100  # Set progress to 100% to indicate completion (even though it's an error)
            )
        return None
```

## Tracking Extraction Status

The application tracks the status of the extraction process using a cache-based system:

```python
def update_extraction_status(user_id, step=None, status=None, message=None, progress=None, extracted_data=None, overall_status=None):
    """Update the extraction status for a user"""
    cache_key = get_extraction_cache_key(user_id)
    status_data = get_extraction_status(user_id)
    
    if step and status:
        status_data["steps"][step] = {
            "status": status,
            "message": message
        }
        
        # Update progress based on completed steps
        completed_steps = sum(1 for step_data in status_data["steps"].values() 
                             if step_data["status"] == ExtractionStatus.COMPLETED)
        status_data["progress"] = int((completed_steps / len(status_data["steps"])) * 100)
    
    if progress is not None:
        status_data["progress"] = progress
        
    if message is not None:
        status_data["message"] = message
        
    if extracted_data is not None:
        status_data["extracted_data"] = extracted_data
        
    if overall_status is not None:
        status_data["overall_status"] = overall_status
    
    # Save the updated status
    cache.set(cache_key, status_data, CACHE_TIMEOUT)
    
    return status_data
```

The extraction process is initiated from the job seeker profile page:

@login_required
def extract_resume_data(request):
    """Extract data from resume using AI"""
    import time
    import logging
    import threading

    logger = logging.getLogger(__name__)

    # Check if user is a job seeker
    try:
        user_type = UserType.objects.get(user=request.user)
        if not user_type.is_job_seeker:
            return redirect('dashboard')
    except UserType.DoesNotExist:
        pass

    # For AJAX requests to check status
    if request.headers.get('X-Requested-With') == 'XMLHttpRequest' and request.method == 'GET':
        # Return the current extraction status
        status_data = get_extraction_status(request.user.id)
        return JsonResponse(status_data)

    # Get job seeker profile
    try:
        job_seeker_profile = JobSeekerProfile.objects.get(user=request.user)

        # Check if resume exists
        if not job_seeker_profile.resume:
            messages.error(request, "Please upload a resume before using the AI extraction feature.")
            return redirect('job_seeker_profile')

        # Define a function to run the extraction process in a background thread
        def run_extraction_process():
            try:
                # Call the extraction function with user_id for status tracking
                resume_path = str(job_seeker_profile.resume)
                extracted_data = extract_resume_data_from_api(resume_path, user_id=request.user.id)

                if not extracted_data:
                    logger.error("Resume data extraction failed")
                    return

                # Update the profile with extracted data
                if 'skills' in extracted_data and extracted_data['skills']:
                    job_seeker_profile.skills = extracted_data['skills']

                if 'education' in extracted_data and extracted_data['education']:
                    job_seeker_profile.education = extracted_data['education']

                if 'experience_years' in extracted_data and extracted_data['experience_years']:
                    job_seeker_profile.experience_years = extracted_data['experience_years']

                if 'location' in extracted_data and extracted_data['location']:
                    job_seeker_profile.location = extracted_data['location']

                job_seeker_profile.save()

            except Exception as e:
                logger.error(f"Error in extraction thread: {str(e)}")

        # Start the extraction process in a background thread
        extraction_thread = threading.Thread(target=run_extraction_process)
        extraction_thread.daemon = True
        extraction_thread.start()

        # If this is an AJAX request, return a processing status
        if request.headers.get('X-Requested-With') == 'XMLHttpRequest':
            return JsonResponse({
                'status': 'processing',
                'message': 'Resume extraction has started. Please wait...'
            })

        # For non-AJAX requests, redirect to the profile page with a message
        messages.info(request, "Resume extraction has started. This may take a minute...")
        return redirect('job_seeker_profile')

    except JobSeekerProfile.DoesNotExist:
        messages.error(request, "Profile not found.")
        return redirect('job_seeker_profile')
    except Exception as e:
        logger.error(f"Error in extract_resume_data view: {str(e)}")
        messages.error(request, f"Error extracting data: {str(e)}")
        return redirect('job_seeker_profile')
